  # B2. Funcion que escribe una variable sin dejar rastro en el historial ni en
  #     pantalla. `read -s` no hace eco. Pegala tal cual en la terminal.
  set_env() {
    local key="$1" file="/srv/motomas/CRM-MotoMas-Demo/.env" val
    printf 'Pega el valor de %s (no se vera nada al pegar): ' "$key" >&2
    read -rs val; echo >&2
    [ -z "$val" ] && { echo "vacio, no se toca $key" >&2; return 1; }
    if grep -qE "^${key}=" "$file"; then
      python3 - "$file" "$key" "$val" <<'PY'
import sys,io,re
f,k,v=sys.argv[1],sys.argv[2],sys.argv[3]
s=io.open(f,encoding="utf-8").read()
s=re.sub(r'(?m)^'+re.escape(k)+r'=.*$', k+'="'+v+'"', s, count=1)
io.open(f,"w",encoding="utf-8").write(s)
PY
    else
      printf '%s="%s"\n' "$key" "$val" >> "$file"
    fi
    echo "$key escrito (${#val} caracteres)" >&2
  }
